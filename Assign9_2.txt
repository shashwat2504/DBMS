mysql> create procedure tax()
    -> begin
    -> declare done int default 0;
    -> declare a_name varchar(20);
    -> declare a_price int;
    -> declare a_city varchar(20);
    -> declare a_state varchar(2);
    -> declare a_rate decimal(5,2);
    -> declare tax decimal(10,2);
    -> declare taxcursor cursor for select c_name,price_of_product,city,state from customer;
    -> declare continue handler for not found set done=1;
    -> open taxcursor;
    -> repeat
    -> fetch taxcursor into a_name,a_price,a_city,a_state;
    -> if not done then
    -> select tax_rate into a_rate from sales_tax_rate where state=a_state;
    -> set tax=a_price*a_rate/100;
    -> insert into tax_final values(a_name,a_city,a_state,tax);
    -> end if;
    -> until done end repeat;
    -> close taxcursor;
    -> end//
